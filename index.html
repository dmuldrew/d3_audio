Quickstart Example

import * as d3 from "d3"; import { scalePitch, scaleGain, scalePan, timeline, choreography } from "d3-audio"; // 1. Define D3-like audio scalers const pitchScale = scalePitch() .domain([0, 100]) .range(["C3", "C6"]) .scale("pentatonic") .root("C"); const gainScale = scaleGain().domain([0, 50]).range([0.2, 1.0]); const panScale = scalePan().domain([0, 800]).range([-1, 1]); // 2. Connect dataset to Tone.js Timeline const tl = timeline({ bpm: 120, loop: true }) .data(dataset) .time((d, i) => i * 0.25) .pitch(d => pitchScale(d.value)) .gain(d => gainScale(d.metric)) .pan(d => panScale(d.x)) .movement(d => ({ movement: d.category === "alert" ? "wiggle" : "flip", intensity: d.value / 50, element: d3.select(`#node-\${d.id}`) })); // 3. Play synchronized audio + visual choreography await tl.play();
